if(RFC_ENABLE_VULKAN)
  find_package(Vulkan 1.1 REQUIRED)
  find_program(RFC_GLSLANG_VALIDATOR glslangValidator REQUIRED)
  set(rfc_spirv_dir "${CMAKE_CURRENT_BINARY_DIR}/generated/vulkan")
  file(MAKE_DIRECTORY "${rfc_spirv_dir}")
  add_custom_command(OUTPUT "${rfc_spirv_dir}/packed_binary.spv"
    COMMAND "${RFC_GLSLANG_VALIDATOR}" -V --target-env vulkan1.1
      -o "${rfc_spirv_dir}/packed_binary.spv" "${CMAKE_CURRENT_SOURCE_DIR}/src/backend/packed_binary.comp"
    DEPENDS src/backend/packed_binary.comp VERBATIM)
  add_custom_command(OUTPUT "${rfc_spirv_dir}/packed_binary_spirv.hpp"
    COMMAND ${CMAKE_COMMAND} "-DINPUT=${rfc_spirv_dir}/packed_binary.spv"
      "-DOUTPUT=${rfc_spirv_dir}/packed_binary_spirv.hpp" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/EmbedSpirv.cmake"
    DEPENDS "${rfc_spirv_dir}/packed_binary.spv" cmake/EmbedSpirv.cmake VERBATIM)
  add_library(rfc_vulkan_binary STATIC src/backend/vulkan_binary.cpp "${rfc_spirv_dir}/packed_binary_spirv.hpp")
  add_library(rfc::vulkan_binary ALIAS rfc_vulkan_binary)
  set_target_properties(rfc_vulkan_binary PROPERTIES EXPORT_NAME vulkan_binary POSITION_INDEPENDENT_CODE ON)
  target_include_directories(rfc_vulkan_binary PRIVATE "${rfc_spirv_dir}")
  target_link_libraries(rfc_vulkan_binary PUBLIC rfc::cpu_tiles rfc::hardware_inventory PRIVATE Vulkan::Vulkan)
  rfc_interface_target(vulkan_work)
  target_link_libraries(rfc_vulkan_work INTERFACE rfc::vulkan_binary rfc::field_work rfc::binary_work)
endif()
