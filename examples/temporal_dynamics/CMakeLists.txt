cmake_minimum_required(VERSION 3.24)
project(rfc_temporal_dynamics LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This demo includes the qualified Linux durable native store")
endif()
if(RFC_SOURCE_DIR)
  foreach(option RFC_BUILD_APPS RFC_BUILD_LOCAL_TOOLS RFC_BUILD_TESTS RFC_BUILD_BENCHMARKS RFC_BUILD_CAPABILITY_DEMO RFC_BUILD_LEARNING)
    set(${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory("${RFC_SOURCE_DIR}" rfc EXCLUDE_FROM_ALL)
else()
  find_package(RFC CONFIG REQUIRED)
endif()
add_executable(temporal_dynamics main.cpp database.cpp fields.cpp algebra.cpp products.cpp)
target_link_libraries(temporal_dynamics PRIVATE rfc::inference)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME temporal_dynamics_exact
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify.py"
      --exe $<TARGET_FILE:temporal_dynamics> --events "${CMAKE_CURRENT_SOURCE_DIR}/events.tsv")
  set_tests_properties(temporal_dynamics_exact PROPERTIES TIMEOUT 120
    ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
