cmake_minimum_required(VERSION 3.24)
project(rfc_algebraic_database_example LANGUAGES CXX)
if(RFC_SOURCE_DIR)
  foreach(option RFC_BUILD_APPS RFC_BUILD_LOCAL_TOOLS RFC_BUILD_TESTS RFC_BUILD_BENCHMARKS
                 RFC_BUILD_CAPABILITY_DEMO RFC_BUILD_LEARNING)
    set(${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory("${RFC_SOURCE_DIR}" rfc EXCLUDE_FROM_ALL)
else()
  find_package(RFC CONFIG REQUIRED)
endif()
add_executable(algebraic_database main.cpp)
target_link_libraries(algebraic_database PRIVATE rfc::native_query_execution)
add_executable(algebraic_seed seed.cpp)
target_link_libraries(algebraic_seed PRIVATE rfc::storage)
add_executable(algebraic_query_cell query.cpp)
target_link_libraries(algebraic_query_cell PRIVATE rfc::native_query_execution)
