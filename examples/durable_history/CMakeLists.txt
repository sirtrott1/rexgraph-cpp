cmake_minimum_required(VERSION 3.24)
project(rfc_durable_history_consumer LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This durable storage profile requires Linux")
endif()
get_filename_component(RFC_DEFAULT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(RFC_SOURCE_DIR "${RFC_DEFAULT_SOURCE}" CACHE PATH "RexGraphCPP source checkout")
set(RFC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RFC_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(RFC_BUILD_LOCAL_TOOLS OFF CACHE BOOL "" FORCE)
set(RFC_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${RFC_SOURCE_DIR}" rfc EXCLUDE_FROM_ALL)
add_executable(durable_history main.cpp)
target_link_libraries(durable_history PRIVATE rfc::storage)
