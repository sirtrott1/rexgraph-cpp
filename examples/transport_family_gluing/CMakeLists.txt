cmake_minimum_required(VERSION 3.24)
project(rfc_transport_family_gluing_example LANGUAGES CXX)
option(RFC_BUILD_LEARNING "Build the transport family learning example" ON)
if(NOT RFC_BUILD_LEARNING)
  return()
endif()
if(RFC_SOURCE_DIR)
  foreach(option RFC_BUILD_APPS RFC_BUILD_LOCAL_TOOLS RFC_BUILD_TESTS RFC_BUILD_BENCHMARKS
                 RFC_BUILD_CAPABILITY_DEMO)
    set(${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory("${RFC_SOURCE_DIR}" rfc EXCLUDE_FROM_ALL)
else()
  find_package(RFC CONFIG REQUIRED)
endif()
find_package(Threads REQUIRED)
add_executable(transport_family_gluing main.cpp)
target_link_libraries(transport_family_gluing PRIVATE rfc::learning Threads::Threads)
