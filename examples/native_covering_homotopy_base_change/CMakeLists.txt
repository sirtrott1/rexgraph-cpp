cmake_minimum_required(VERSION 3.20)
project(native_covering_homotopy_base_change LANGUAGES CXX)
if(RFC_SOURCE_DIR)
  set(RFC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_APPS OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_LOCAL_TOOLS OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_CAPABILITY_DEMO OFF CACHE BOOL "" FORCE)
  add_subdirectory("${RFC_SOURCE_DIR}" rfc)
else()
  find_package(RFC CONFIG REQUIRED)
endif()
add_executable(native_covering_homotopy_base_change main.cpp)
target_link_libraries(native_covering_homotopy_base_change PRIVATE rfc::inference)
