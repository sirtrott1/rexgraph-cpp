cmake_minimum_required(VERSION 3.24)
project(rfc_database_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(RFC_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Pinned RFC library source")
foreach(opt RFC_BUILD_APPS RFC_BUILD_LOCAL_TOOLS RFC_BUILD_TESTS RFC_BUILD_BENCHMARKS RFC_BUILD_CAPABILITY_DEMO RFC_BUILD_LEARNING BUILD_TESTING)
 set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${RFC_SOURCE_DIR}" rfc EXCLUDE_FROM_ALL)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(database_preview INTERFACE)
target_include_directories(database_preview INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(database_preview INTERFACE rfc::native_execution OpenSSL::Crypto)
if(NOT UNIX)
 message(FATAL_ERROR "This preview's filesystem backend is POSIX-only")
endif()
add_executable(rfcdb main.cpp)
target_link_libraries(rfcdb PRIVATE database_preview)
target_compile_options(rfcdb PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(database_tests tests/test_database.cpp)
target_link_libraries(database_tests PRIVATE database_preview)
target_compile_options(database_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(DB_SANITIZERS "ASan and UBSan" OFF)
if(DB_SANITIZERS)
 foreach(t rfcdb database_tests)
  target_compile_options(${t} PRIVATE -O1 -g0 -fno-omit-frame-pointer -fsanitize=address,undefined)
  target_link_options(${t} PRIVATE -fsanitize=address,undefined)
 endforeach()
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME database_roundtrip COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_cpp.py $<TARGET_FILE:database_tests>)
add_test(NAME database_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:rfcdb>)
install(TARGETS rfcdb RUNTIME DESTINATION bin)
