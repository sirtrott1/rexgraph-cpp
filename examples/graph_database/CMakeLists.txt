cmake_minimum_required(VERSION 3.20)
project(rfc_graph_database_consumer LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This durable consumer uses the qualified exclusive Linux store")
endif()
if(RFC_SOURCE_DIR)
  set(RFC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_APPS OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_LOCAL_TOOLS OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_CAPABILITY_DEMO OFF CACHE BOOL "" FORCE)
  set(RFC_BUILD_LEARNING OFF CACHE BOOL "" FORCE)
  set(RFC_ENABLE_SQLITE_INTERPRETATION OFF CACHE BOOL "" FORCE)
  set(RFC_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
  add_subdirectory("${RFC_SOURCE_DIR}" rfc EXCLUDE_FROM_ALL)
else()
  find_package(RFC CONFIG REQUIRED)
endif()
add_executable(graph_database main.cpp)
target_link_libraries(graph_database PRIVATE rfc::native_query_execution)
