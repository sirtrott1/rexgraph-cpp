cmake_minimum_required(VERSION 3.24)
project(rfc_batch_harmonic_attachment_example LANGUAGES CXX)
if(RFC_SOURCE_DIR)
  foreach(option RFC_BUILD_APPS RFC_BUILD_LOCAL_TOOLS RFC_BUILD_TESTS RFC_BUILD_BENCHMARKS
                 RFC_BUILD_CAPABILITY_DEMO RFC_BUILD_LEARNING)
    set(${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory("${RFC_SOURCE_DIR}" rfc EXCLUDE_FROM_ALL)
else()
  find_package(RFC CONFIG REQUIRED)
endif()
find_package(Threads REQUIRED)
add_executable(batch_harmonic_attachment main.cpp)
target_link_libraries(batch_harmonic_attachment PRIVATE rfc::inference Threads::Threads)
