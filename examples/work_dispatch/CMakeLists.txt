cmake_minimum_required(VERSION 3.24)
project(rfc_work_dispatch_consumer LANGUAGES CXX)
set(RFC_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(RFC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RFC_BUILD_LOCAL_TOOLS OFF CACHE BOOL "" FORCE)
set(RFC_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. rfc-build)
add_executable(work_dispatch main.cpp)
target_link_libraries(work_dispatch PRIVATE rfc::native_locality rfc::binary_work)
if(RFC_ENABLE_VULKAN)
  target_link_libraries(work_dispatch PRIVATE rfc::vulkan_work)
  target_compile_definitions(work_dispatch PRIVATE RFC_WORK_VULKAN=1)
endif()
