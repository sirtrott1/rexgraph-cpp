cmake_minimum_required(VERSION 3.24)
project(RFCHomologyDiagramContracts LANGUAGES CXX)
find_package(RFC CONFIG REQUIRED)
enable_testing()
foreach(name rational_zigzag finite_chain_diagram)
  add_executable(${name} ../topology/test_${name}.cpp)
  target_link_libraries(${name} PRIVATE rfc::topology)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ${name} COMMAND ${name})
endforeach()
add_executable(native_homology_diagram ../inference/test_native_homology_diagram.cpp)
target_link_libraries(native_homology_diagram PRIVATE rfc::inference)
target_compile_options(native_homology_diagram PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME native_homology_diagram COMMAND native_homology_diagram)
add_executable(zigzag_consumer ../topology/zigzag_consumer.cpp)
target_link_libraries(zigzag_consumer PRIVATE rfc::topology)
target_compile_options(zigzag_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
